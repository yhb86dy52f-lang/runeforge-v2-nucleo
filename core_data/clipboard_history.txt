; ========= TELEMETRÍA (WMI) =========
GetTelemetry() {
    static wmi := ComObjGet("winmgmts:\\.\root\cimv2")
    cpu := wmi.ExecQuery("SELECT LoadPercentage FROM Win32_Processor").ItemIndex(0).LoadPercentage
    os := wmi.ExecQuery("SELECT FreePhysicalMemory, TotalVisibleMemorySize FROM Win32_OperatingSystem").ItemIndex(0)
    ramUsed := (os.TotalVisibleMemorySize - os.FreePhysicalMemory) / 1024 / 1024
    return { cpu: cpu, ram: Format("{:.1f}", ramUsed) }
}

; ========= LÓGICA DE ESTADO =========
IsPortListening(port) {
    static wsadata := Buffer(400, 0)
    if (DllCall("ws2_32\WSAStartup", "UShort", 0x0202, "Ptr", wsadata) != 0)
        return false
    sock := DllCall("ws2_32\socket", "Int", 2, "Int", 1, "Int", 6, "Ptr")
    if (sock = -1) {
        DllCall("ws2_32\WSACleanup")
        return false
    }
    addr := Buffer(16, 0)
    NumPut("UShort", 2, addr, 0)
    NumPut("UShort", DllCall("ws2_32\htons", "UShort", port, "UShort"), addr, 2)
    NumPut("UInt", DllCall("ws2_32\inet_addr", "AStr", "127.0.0.1", "UInt"), addr, 4)
    res := DllCall("ws2_32\connect", "Ptr", sock, "Ptr", addr, "Int", 16, "Int")
    DllCall("ws2_32\closesocket", "Ptr", sock)
    DllCall("ws2_32\WSACleanup")
    return (res == 0)
}
