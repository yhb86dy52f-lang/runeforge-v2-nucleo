$retrievePy = @'
import sys
import os
import argparse
import pickle
import numpy as np

os.environ['TF_CPP_MIN_LOG_LEVEL'] = '3'
try:
    import faiss
    from sentence_transformers import SentenceTransformer
except ImportError:
    sys.exit(0)

BASE_DIR = os.path.dirname(__file__)
KB_FILE = os.path.join(BASE_DIR, 'knowledge_base.txt')
INDEX_FILE = os.path.join(BASE_DIR, 'faiss_index.bin')
CHUNKS_FILE = os.path.join(BASE_DIR, 'chunks.pkl')

EMBEDDING_MODEL = 'all-MiniLM-L6-v2'
CHUNK_SIZE = 400

def chunk_text(text, size):
    words = text.split()
    chunks, current, length = [], [], 0
    for w in words:
        if length + len(w) > size:
            chunks.append(" ".join(current))
            current = [w]
            length = len(w)
        else:
            current.append(w)
            length += len(w) + 1
    if current:
        chunks.append(" ".join(current))
    return chunks

def load_or_create_index():
    original_stderr = sys.stderr
    sys.stderr = open(os.devnull, 'w')
    model = SentenceTransformer(EMBEDDING_MODEL)
    sys.stderr = original_stderr

    if os.path.exists(INDEX_FILE) and os.path.exists(CHUNKS_FILE):
        index = faiss.read_index(INDEX_FILE)
        with open(CHUNKS_FILE, 'rb') as f:
            chunks = pickle.load(f)
        return model, index, chunks

    if not os.path.exists(KB_FILE):
        return model, None, []

    with open(KB_FILE, 'r', encoding='utf-8') as f:
        raw = f.read()
    chunks = chunk_text(raw, CHUNK_SIZE)
    
    original_stderr = sys.stderr
    sys.stderr = open(os.devnull, 'w')
    emb = model.encode(chunks, convert_to_numpy=True)
    sys.stderr = original_stderr

    index = faiss.IndexFlatL2(emb.shape[1])
    index.add(emb)
    faiss.write_index(index, INDEX_FILE)
    with open(CHUNKS_FILE, 'wb') as f:
        pickle.dump(chunks, f)
    return model, index, chunks

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('query', nargs='*')
    parser.add_argument('--top_k', type=int, default=3)
    args = parser.parse_args()
    query = ' '.join(args.query).strip()
    if not query:
        sys.exit(0)

    model, index, chunks = load_or_create_index()
    if index is None or index.ntotal == 0:
        sys.exit(0)

    q_vec = model.encode([query], convert_to_numpy=True)
    k = min(args.top_k, index.ntotal)
    _, indices = index.search(q_vec, k)
    
    res = [chunks[i] for i in indices[0] if i != -1 and i < len(chunks)]
    sys.stdout.write("\n---\n".join(res))

if __name__ == '__main__':
    main()
